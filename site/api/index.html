    class AsyncTask:
    """
    Représente une tâche de calcul asynchrone (Kernel JIT) qui peut être exécutée
    sur n'importe quel périphérique disponible (CUDA, OpenCL, LLVM ou LLVM NVPTX).
    """

    def __init__(
        self,
        kernel_code_opencl: str,
        kernel_name_opencl: str,
        kernel_code_cuda: str,
        kernel_name_cuda: str,
        kernel_code_llvm: str | None = None,
        kernel_name_llvm: str | None = None,
        kernel_code_ptx: str | None = None,
        kernel_name_ptx: str | None = None,
    ):
        self.kernel_code_opencl = kernel_code_opencl
        self.kernel_name_opencl = kernel_name_opencl
        self.kernel_code_cuda = kernel_code_cuda
        self.kernel_name_cuda = kernel_name_cuda
        self.kernel_code_llvm = kernel_code_llvm
        self.kernel_name_llvm = kernel_name_llvm
        self.kernel_code_ptx = kernel_code_ptx
        self.kernel_name_ptx = kernel_name_ptx
        self._compiled_programs: dict = {}

    async def execute(self, scheduler, global_size, local_size, *args, stream=None, fire_and_forget=False):
        """
        Orchestre l'exécution du kernel en déléguant l'exécution matérielle
        directement au manager du périphérique cible.
        """
        if not args:
            device_manager = scheduler.get_best_device()
        else:
            device_manager = args[0].device_manager

        return await device_manager.execute_kernel(
            self, global_size, local_size, *args, stream=stream, fire_and_forget=fire_and_forget
        )

    class AsyncBuffer:
    """
    Représente un buffer de données stocké en mémoire physique GPU (VRAM).
    Cette classe est générique et délègue toutes les opérations d'allocation,
    de lecture, et d'écriture au manager de périphérique associé.
    """

    def __init__(self, device_manager, buffer_obj, shape, dtype, backend="opencl"):
        self.device_manager = device_manager
        self.cl_buffer = buffer_obj  # cupy.ndarray (CUDA) ou pyopencl.Buffer (OpenCL)
        self.shape = shape
        self.dtype = dtype
        self.backend = backend

    def __del__(self):
        """
        Retourne automatiquement le buffer physique au pool du manager de périphérique.
        """
        if hasattr(self, "device_manager") and self.device_manager is not None:
            if hasattr(self.device_manager, "free_buffer"):
                try:
                    self.device_manager.free_buffer(self.cl_buffer, self.shape, self.dtype)
                except Exception:
                    pass

    @classmethod
    @auto_sync
    async def to_device(cls, scheduler=None, host_data: np.ndarray | None = None):
        """
        Copie asynchrone et non-bloquante de la RAM de l'hôte vers la VRAM du GPU.
        """
        if host_data is None and isinstance(scheduler, np.ndarray):
            # Signature backward-compatibility: si l'appel était to_device(host_data)
            host_data = scheduler
            scheduler = None

        if scheduler is None:
            from .scheduler import get_default_scheduler

            scheduler = get_default_scheduler()

        if hasattr(scheduler, "get_queue"):
            device_manager = scheduler
        else:
            device_manager = scheduler.get_best_device()

        if host_data is None:
            raise ValueError("host_data ne peut pas être None dans to_device")

        buffer_obj = await device_manager.allocate_buffer_from_host(host_data)
        return cls(device_manager, buffer_obj, host_data.shape, host_data.dtype, backend=device_manager.backend)

    @auto_sync
    async def overwrite(self, host_data: np.ndarray):
        """
        Écrit les données de l'hôte dans un buffer GPU existant sans réallouer.
        """
        await self.device_manager.write_buffer(self.cl_buffer, host_data)

    @classmethod
    def empty_like(cls, scheduler=None, host_data: np.ndarray | None = None):
        """
        Alloue un buffer vide en VRAM de même taille et type que host_data.
        """
        if host_data is None and isinstance(scheduler, np.ndarray):
            host_data = scheduler
            scheduler = None

        if scheduler is None:
            from .scheduler import get_default_scheduler

            scheduler = get_default_scheduler()

        if hasattr(scheduler, "get_queue"):
            device_manager = scheduler
        else:
            device_manager = scheduler.get_best_device()

        if host_data is None:
            raise ValueError("host_data ne peut pas être None dans empty_like")

        buffer_obj = device_manager.allocate_empty_buffer(host_data.shape, host_data.dtype)
        return cls(device_manager, buffer_obj, host_data.shape, host_data.dtype, backend=device_manager.backend)

    @auto_sync
    async def to_host(self, out: np.ndarray | None = None) -> Any:
        """
        Copie non-bloquante de la VRAM du GPU vers la RAM de l'hôte.
        """
        return await self.device_manager.read_buffer(self.cl_buffer, self.shape, self.dtype, out=out)

    @auto_sync
    async def dirty_read(self, stream=None) -> Any:
        """
        Lecture rapide du buffer (DMA page-locked/command queue séparée) sans bloquer
        le stream de travail principal.
        """
        return await self.device_manager.dirty_read_buffer(self.cl_buffer, self.shape, self.dtype, stream=stream)

    def __dlpack__(self, stream=None):
        """
        Exporte le buffer sous forme de PyCapsule DLPack (Zero-Copy VRAM).
        Compatible avec PyTorch (torch.from_dlpack), CuPy (cupy.from_dlpack), etc.
        """
        if hasattr(self.cl_buffer, "__dlpack__"):
            try:
                return self.cl_buffer.__dlpack__(stream=stream)
            except TypeError:
                return self.cl_buffer.__dlpack__()
        elif hasattr(self.cl_buffer, "toDlpack"):
            return self.cl_buffer.toDlpack()
        else:
            raise NotImplementedError(
                f"L'objet buffer sous-jacent ({type(self.cl_buffer).__name__}) sous le backend '{self.backend}' "
                "ne prend pas en charge l'exportation DLPack directe."
            )

    def __dlpack_device__(self):
        """
        Renvoie la description du périphérique au format DLPack : (device_type, device_id).
        Ex: (1, 0) pour kDLCUDA (device 0), (13, 0) pour kDLCPU, (3, 0) pour kDLOpenCL.
        """
        if hasattr(self.cl_buffer, "__dlpack_device__"):
            return self.cl_buffer.__dlpack_device__()

        if self.backend == "cuda":
            dev_id = getattr(getattr(self.device_manager, "device", None), "id", 0)
            return (1, dev_id)  # 1 = kDLCUDA
        elif self.backend == "opencl":
            return (3, 0)  # 3 = kDLOpenCL
        else:
            return (13, 0)  # 13 = kDLCPU

    @classmethod
    def from_dlpack(cls, dlpack_obj, scheduler=None):
        """
        Crée un AsyncBuffer à partir d'un objet externe compatible DLPack (ex: PyTorch Tensor, CuPy array)
        sans aucune copie mémoire (Zero-Copy).
        """
        if scheduler is not None:
            if hasattr(scheduler, "get_queue"):
                device_manager = scheduler
            else:
                device_manager = scheduler.get_best_device()
        else:
            device_manager = None

        buffer_obj = None
        backend = "cuda"

        try:
            # pyrefly: ignore [missing-import]
            import cupy as cp

            if isinstance(dlpack_obj, cp.ndarray):
                buffer_obj = dlpack_obj
            else:
                buffer_obj = cp.from_dlpack(dlpack_obj)
            backend = "cuda"
        except Exception:
            buffer_obj = dlpack_obj
            backend = getattr(dlpack_obj, "backend", "cuda")

        shape = getattr(buffer_obj, "shape", ())
        dtype = getattr(buffer_obj, "dtype", np.float32)

        return cls(device_manager, buffer_obj, shape, dtype, backend=backend)

    class Scheduler:
    def __init__(self, num_queues_per_device: int = 4, backend: str | None = None, gpu: Any = None):
        """
        Initialise un Scheduler.

        Args:
            num_queues_per_device (int): Nombre de files de commande (streams/queues) par GPU.
            backend (str, optional): "opencl" ou "cuda". Si None et gpu est fourni, on utilise le backend du gpu.
            gpu (GPUDeviceInfo ou raw device, optional): GPU spécifique à utiliser.
        """
        self.gpu = gpu
        if gpu is not None and hasattr(gpu, "backend"):
            self.backend = backend.lower() if backend is not None else gpu.backend
        else:
            self.backend = backend.lower() if backend is not None else "opencl"

        self.devices_managers: List[Any] = []
        self.num_queues_per_device = num_queues_per_device
        self._next_device_idx = 0

    def initialize(self, device_name_filter: str | None = None):
        """
        Initialise le(s) périphérique(s) de calcul.
        Si un `gpu` spécifique a été passé dans le constructeur, il est configuré directement.
        Sinon, effectue une découverte automatique filtrée par `device_name_filter`.
        """
        from .devices import ALL_DEVICE_MANAGERS
        from typing import Any

        lookup_backend = "cuda" if self.backend == "llvm_cuda" else self.backend
        manager_cls: Any = next((c for c in ALL_DEVICE_MANAGERS if getattr(c, "backend", None) == lookup_backend), None)

        if manager_cls is None:
            raise ValueError(
                f"Backend '{self.backend}' non supporté. Backends disponibles: {[getattr(c, 'backend', None) for c in ALL_DEVICE_MANAGERS]}"
            )

        if self.gpu is not None:
            # 1. Utilisation du GPU spécifié
            raw_device = getattr(self.gpu, "raw_device", self.gpu)
            if hasattr(self.gpu, "backend") and self.gpu.backend != lookup_backend:
                logger.warning(
                    f"GPU backend mismatch: specified {self.gpu.backend} but requested {self.backend}. Attempting to resolve."
                )
                devs = manager_cls.discover_devices()
                if self.gpu.index < len(devs):
                    raw_device = devs[self.gpu.index]
                elif len(devs) > 0:
                    raw_device = devs[0]
                else:
                    raise ValueError(
                        f"Specified GPU has backend '{self.gpu.backend}', but requested backend '{self.backend}' has no compatible devices."
                    )

            if lookup_backend == "cuda" and isinstance(raw_device, int):
                # pyrefly: ignore [missing-import]
                import cupy as cp

                raw_device = cp.cuda.Device(raw_device)

            manager = manager_cls(raw_device, num_queues=self.num_queues_per_device)
            manager.backend = self.backend
            self.devices_managers.append(manager)
            logger.info(f"Périphérique {self.backend.upper()} initialisé (spécifié) : {manager.name}")
        else:
            # 2. Découverte automatique dynamique
            devs = manager_cls.discover_devices(name_filter=device_name_filter)
            for dev in devs:
                try:
                    manager = manager_cls(dev, num_queues=self.num_queues_per_device)
                    manager.backend = self.backend
                    self.devices_managers.append(manager)
                    logger.info(f"Périphérique {self.backend.upper()} initialisé : {manager.name}")

                    dev_name = getattr(dev, "name", dev)
                    logger.error(f"Impossible d'initialiser le GPU {self.backend.upper()} {dev_name}: {e}")

                raise RuntimeError(f"Le Scheduler {self.backend.upper()} n'a pu initialiser aucun périphérique.")

        )

    def get_best_device(self):
        """
        Répartit la charge entre les différents GPUs physiques détectés (Round-Robin).
        """
        if not self.devices_managers:
            raise RuntimeError("Scheduler non initialisé ou aucun périphérique détecté.")

        manager = self.devices_managers[self._next_device_idx]
        self._next_device_idx = (self._next_device_idx + 1) % len(self.devices_managers)
        return manager

    def initialize(self, device_name_filter: str | None = None):
    """
    Initialise le(s) périphérique(s) de calcul.
    Si un `gpu` spécifique a été passé dans le constructeur, il est configuré directement.
    Sinon, effectue une découverte automatique filtrée par `device_name_filter`.
    """
    from .devices import ALL_DEVICE_MANAGERS
    from typing import Any

    lookup_backend = "cuda" if self.backend == "llvm_cuda" else self.backend
    manager_cls: Any = next((c for c in ALL_DEVICE_MANAGERS if getattr(c, "backend", None) == lookup_backend), None)

    if manager_cls is None:
        raise ValueError(
            f"Backend '{self.backend}' non supporté. Backends disponibles: {[getattr(c, 'backend', None) for c in ALL_DEVICE_MANAGERS]}"
        )

    if self.gpu is not None:
        # 1. Utilisation du GPU spécifié
        raw_device = getattr(self.gpu, "raw_device", self.gpu)
        if hasattr(self.gpu, "backend") and self.gpu.backend != lookup_backend:
            logger.warning(
                f"GPU backend mismatch: specified {self.gpu.backend} but requested {self.backend}. Attempting to resolve."
            )
            devs = manager_cls.discover_devices()
            if self.gpu.index < len(devs):
                raw_device = devs[self.gpu.index]
            elif len(devs) > 0:
                raw_device = devs[0]
            else:
                raise ValueError(
                    f"Specified GPU has backend '{self.gpu.backend}', but requested backend '{self.backend}' has no compatible devices."
                )

        if lookup_backend == "cuda" and isinstance(raw_device, int):
            # pyrefly: ignore [missing-import]
            import cupy as cp

            raw_device = cp.cuda.Device(raw_device)

        manager = manager_cls(raw_device, num_queues=self.num_queues_per_device)
        manager.backend = self.backend
        self.devices_managers.append(manager)
        logger.info(f"Périphérique {self.backend.upper()} initialisé (spécifié) : {manager.name}")
    else:
        # 2. Découverte automatique dynamique
        devs = manager_cls.discover_devices(name_filter=device_name_filter)
        for dev in devs:
            try:
                manager = manager_cls(dev, num_queues=self.num_queues_per_device)
                manager.backend = self.backend
                self.devices_managers.append(manager)
                logger.info(f"Périphérique {self.backend.upper()} initialisé : {manager.name}")
            except Exception as e:
                dev_name = getattr(dev, "name", dev)
                logger.error(f"Impossible d'initialiser le GPU {self.backend.upper()} {dev_name}: {e}")

        if not self.devices_managers:
            raise RuntimeError(f"Le Scheduler {self.backend.upper()} n'a pu initialiser aucun périphérique.")

    logger.info(
        f"Scheduler {self.backend.upper()} prêt. {len(self.devices_managers)} périphérique(s) disponible(s)."
    )

    class SwarmScheduler:
    """
    Le Quartier Général pour l'architecture ABA / Swarm.
    Initialise automatiquement CUDA pour les cartes Nvidia, et OpenCL pour le reste (Intel, AMD).
    Unifie tous les managers de périphériques en un seul pool de calcul.
    """

    def __init__(self, num_queues_per_device: int = 4, gpus: List[Any] | None = None):
        """
        Initialise un SwarmScheduler.

        Args:
            num_queues_per_device (int): Nombre de files de commande par GPU.
            gpus (List, optional): Liste restreinte de périphériques (GPUDeviceInfo ou bruts) à ajouter à l'essaim.
                                   Si None, découvre automatiquement tous les GPUs du système.
        """
        self.num_queues_per_device = num_queues_per_device
        self.gpus = gpus
        self.devices_managers: List[Any] = []
        self._next_device_idx = 0

    def initialize(self):
        from .devices import ALL_DEVICE_MANAGERS

        # 1. Si une liste spécifique de périphériques est fournie
        if self.gpus is not None:
            for gpu in self.gpus:
                backend = getattr(gpu, "backend", None)
                raw_device = getattr(gpu, "raw_device", gpu)

                # Détection automatique du backend si non spécifié (heuristic)
                if backend is None:
                    try:
                        # pyrefly: ignore [missing-import]
                        import pyopencl as cl

                        if isinstance(raw_device, cl.Device):
                            backend = "opencl"
                    except Exception:
                        pass
                    if backend is None:
                        backend = "cuda"  # fallback par défaut

                # Recherche du manager de périphérique associé au backend
                manager_cls = next((c for c in ALL_DEVICE_MANAGERS if c.backend == backend), None)
                if manager_cls is None:
                    logger.error(f"[Swarm] Aucun manager trouvé pour le backend '{backend}'")
                    continue

                try:
                    if backend == "cuda" and isinstance(raw_device, int):
                        # pyrefly: ignore [missing-import]
                        import cupy as cp

                        raw_device = cp.cuda.Device(raw_device)
                    manager = manager_cls(raw_device, num_queues=self.num_queues_per_device)
                    self.devices_managers.append(manager)
                    logger.info(f"[Swarm] Périphérique {backend.upper()} initialisé (spécifié) : {manager.name}")
                except Exception as e:
                    logger.error(f"[Swarm] Impossible d'initialiser {backend.upper()} pour {raw_device}: {e}")
        else:
            # 2. Découverte automatique sur l'ensemble des managers enregistrés
            for manager_cls in ALL_DEVICE_MANAGERS:
                devs = manager_cls.discover_devices()
                for dev in devs:
                    # En mode découverte auto sur Swarm, on évite le doublon Nvidia sous OpenCL
                    if manager_cls.backend == "opencl":
                        dev_name = getattr(dev, "name", "")
                        dev_vendor = getattr(dev, "vendor", "")
                        if "nvidia" in dev_name.lower() or "nvidia" in dev_vendor.lower():
                            continue  # Déjà géré par CUDA en principe

                    try:
                        manager = manager_cls(dev, num_queues=self.num_queues_per_device)
                        self.devices_managers.append(manager)
                        logger.info(f"[Swarm] {manager_cls.backend.upper()} détecté et ajouté : {manager.name}")
                    except Exception as e:
                        dev_name = getattr(dev, "name", dev)
                        logger.error(f"[Swarm] Erreur lors de l'initialisation de {dev_name}: {e}")

        if not self.devices_managers:
            raise RuntimeError("SwarmScheduler n'a pu initialiser AUCUN périphérique de calcul !")

        logger.info(f"SwarmScheduler PRÊT. {len(self.devices_managers)} périphérique(s) dans l'essaim.")

    def get_best_device(self):
        if not self.devices_managers:
            raise RuntimeError("SwarmScheduler non initialisé.")
        manager = self.devices_managers[self._next_device_idx]
        self._next_device_idx = (self._next_device_idx + 1) % len(self.devices_managers)
        return manager

    def get_device(self, identifier: str):
        """
        identifier: ex "cuda:0", "opencl:1", ou "intel"
        """
        identifier = identifier.lower()
        if ":" in identifier:
            backend, idx_str = identifier.split(":")
            gpu_idx = int(idx_str)
            # Filtre par backend
            managers = [m for m in self.devices_managers if hasattr(m, "backend") and m.backend == backend]
            if gpu_idx < len(managers):
                return managers[gpu_idx]
        else:
            # Filtre par nom
            for m in self.devices_managers:
                if hasattr(m, "name") and identifier in m.name.lower():
                    return m

        # Fallback si non trouvé
        logger.warning(f"[Swarm] Périphérique '{identifier}' non trouvé, fallback sur Round-Robin.")
        return self.get_best_device()
